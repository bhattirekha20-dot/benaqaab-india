## Exact caption grouping patch

```python
def group_words(alignments, max_words=6, max_seconds=2.4):
    cues, current = [], []
    for word in alignments:
        if current and (len(current) >= max_words or word['end'] - current[0]['start'] > max_seconds):
            cues.append({'start': current[0]['start'], 'end': current[-1]['end'],
                         'text': ' '.join(x['word'] for x in current)})
            current = []
        current.append(word)
    if current:
        cues.append({'start': current[0]['start'], 'end': current[-1]['end'],
                     'text': ' '.join(x['word'] for x in current)})
    return cues
```
